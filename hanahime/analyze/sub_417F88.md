{
    int vertexSize = 16;  // Base size
    
    // Optional single-component flags (4 bytes each)
    if (fvf & D3DFVF_RESERVED0)   vertexSize += 4;
    if (fvf & D3DFVF_PSIZE)       vertexSize += 4;
    if (fvf & D3DFVF_DIFFUSE)     vertexSize += 4;
    if (fvf & D3DFVF_SPECULAR)    vertexSize += 4;
    
    // Normal (3 floats = 12 bytes)
    if (fvf & D3DFVF_NORMAL)      vertexSize += 12;
    
    // Position type (bits 1-3)
    switch (fvf & D3DFVF_POSITION_MASK) {
        case D3DFVF_XYZ:      vertexSize += 12; break;  // 3 floats
        case D3DFVF_XYZRHW:   vertexSize += 16; break;  // 4 floats
        case D3DFVF_XYZB1:    vertexSize += 16; break;  // XYZ + 1 weight
        case D3DFVF_XYZB2:    vertexSize += 20; break;  // XYZ + 2 weights
        case D3DFVF_XYZB3:    vertexSize += 24; break;  // XYZ + 3 weights
        case D3DFVF_XYZB4:    vertexSize += 28; break;  // XYZ + 4 weights
        case D3DFVF_XYZB5:    vertexSize += 32; break;  // XYZ + 5 weights
    }
    
    // Texture coordinates
    int texCount = (fvf & D3DFVF_TEXCOUNT_MASK) >> 8;
    for (int i = 0; i < texCount; i++) {
        int texSize = (fvf >> (16 + 2*i)) & 3;
        vertexSize += 4 * g_texCoordSizeLUT[texSize];
    }
    
    return vertexSize;
}
